cmake_minimum_required(VERSION 3.16)
project(pai_sim_safety)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(pai_diff_drive SHARED src/SafeDiffDrive.cc)
target_compile_features(pai_diff_drive PUBLIC cxx_std_17)
target_include_directories(pai_diff_drive PRIVATE include)
target_link_libraries(pai_diff_drive gz-sim8::gz-sim8 gz-plugin2::register)
install(TARGETS pai_diff_drive LIBRARY DESTINATION lib)
install(FILES UPSTREAM.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(command_lease_test test/command_lease_test.cc)
  target_compile_features(command_lease_test PUBLIC cxx_std_17)
  target_include_directories(command_lease_test PRIVATE include)
  add_test(NAME command_lease_test COMMAND command_lease_test)
endif()
ament_package()
